3.1 Pipeline Specifications

Analysis A: Total Registrations by Category

db.events.aggregate([ { $unwind: "$registrations" }, { $group: { _id: "$category", totalEvents: { $addToSet: "$_id" }, confirmedRegistrations: { $sum: 1 } } }, { $project: { category: "$_id", eventsCount: { $size: "$totalEvents" }, confirmedRegistrations: 1 } }, { $sort: { confirmedRegistrations: -1 } } ]);

Analysis B: Top 5 Popular Events by Occupancy Rate

db.events.aggregate([ { $project: { title: 1, category: 1, capacity: 1, registrationsCount: { $size: { $ifNull: ["$registrations", []] } }, occupancyRate: { $multiply: [ { $divide: [ { $size: { $ifNull: ["$registrations", []] } }, "$capacity" ] }, 100 ] } } }, { $sort: { occupancyRate: -1 } }, { $limit: 5 } ]);

Analysis C: Inactive Users (Zero Registrations)

db.users.aggregate([ { $lookup: { from: "events", localField: "_id", foreignField: "registrations.userId", as: "userEvents" } }, { $match: { userEvents: { $size: 0 } } }, { $project: { firstName: 1, lastName: 1, email: 1, role: 1 } } ]);
